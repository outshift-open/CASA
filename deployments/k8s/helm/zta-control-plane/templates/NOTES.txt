Namespace: {{ .Release.Namespace }}

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Wait for all pods to become ready:

   kubectl wait --for=condition=ready pod \
     -l app.kubernetes.io/instance={{ .Release.Name }} \
     -n {{ .Release.Namespace }} \
     --timeout=300s

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2. Access the Auth Service (API + Swagger UI):

   kubectl port-forward \
     -n {{ .Release.Namespace }} \
     svc/{{ .Release.Name }}-auth-service 8000:8000

   Open: http://localhost:8000/docs

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3. Access the ZTA Explorer UI:

   kubectl port-forward \
     -n {{ .Release.Namespace }} \
     svc/{{ .Release.Name }}-ui-explorer 1234:80

   Open: http://localhost:1234

   NOTE: The UI image must be built with:
   {{- if .Values.uiExplorer.nginx.apiProxyEnabled }}
     VITE_API_BASE_URL=/api   (nginx proxy is enabled — /api/ → auth-service)
   {{- else }}
     VITE_API_BASE_URL=<full auth-service URL>
   {{- end }}

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4. Access Keycloak (admin console):

   kubectl port-forward \
     -n {{ .Release.Namespace }} \
     svc/{{ .Release.Name }}-keycloak 8080:8080

   Open: http://localhost:8080
   User: {{ .Values.keycloak.admin.username }} / <see keycloak-secret>

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
For production use, override the default passwords:

   helm upgrade {{ .Release.Name }} . \
     --set authService.database.password=<strong-password> \
     --set authService.idp.adminPassword=<strong-password> \
     --set postgresAuth.password=<strong-password> \
     --set postgresKeycloak.password=<strong-password> \
     --set keycloak.admin.password=<strong-password>
